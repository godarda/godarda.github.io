#include 
#include 
int main()
{
    char s[50];
    printf("———————————————————————————————————————————");
    printf("\nProgram to print length of a given string");
    printf("\n———————————————————————————————————————————");
    printf("\nEnter the string ");
    scanf("%s", s);
    // strlen excludes the terminating null character from its count.
    int l = strlen(s);
    printf("\nLength of string is %d", l);
    printf("\n———————————————————————————————————————————\n");
    return 0;
}
