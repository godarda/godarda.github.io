// Shows that printf returns a positive character count, affecting each control expression.
#include 
int main()
{
    if (printf("Hello, World!\n"))
    {
    }

    // printf returns a positive character count, so its negation is false here.
    while (!printf("Hello, World!\n"))
    {
    }

    switch (printf("Hello, World!\n"))
    {
    }
}
