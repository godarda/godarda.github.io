// Demonstrates that an inner block's a shadows, but does not replace, the outer a.
#include 
int main()
{
    int a = 20;
    {
        // This declaration shadows the outer a only until the block ends.
        int a = 40;
        // variables can have same name in different scope
    }
    printf("%d\n", a);
    return 0;
}
