// Shows that zero is false and !0 evaluates to true in a condition.
#include 
int main()
{
    int x = 0;
    // Because x is zero, !x evaluates to true.
    if (!x)
        printf("%d\n", !x);
    else
        printf("%d\n", x);
    return 0;
}
