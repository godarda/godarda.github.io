C program to demonstrate the use of break statement
{{ site.cfile }}
{%- highlight c -%}
// Stops the loop when i reaches five, so that value is not printed.
#include 
int main()
{
    for (int i = 0; i < 10; i++)
    {
        // break exits the nearest loop immediately.
        if (i == 5)
        {
            break;
        }
        printf("%d\n", i);
    }
    return 0;
}

C program to demonstrate the use of continue statement
{{ site.cfile }}
{%- highlight c -%}
// Skips only the iteration where i is five; the loop continues afterward.
#include 
int main()
{
    for (int i = 0; i < 10; i++)
    {
        // continue skips the rest of this iteration, then the loop proceeds.
        if (i == 5)
        {
            continue;
        }
        printf("%d\n", i);
    }
    return 0;
}
{%- endhighlight -%}
