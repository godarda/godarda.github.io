// Sums every element of a fixed-size integer array.
#include 
int main()
{
    int sum = 0;
    int arr[5] = {100, 15, 8, 45, 93};
    // The index visits each array element once and adds it to the running total.
    for (int i = 0; i < 5; i++)
    {
        sum = sum + arr[i];
    }
    printf("The addition of a given array is %d ", sum);
    return 0;
}
