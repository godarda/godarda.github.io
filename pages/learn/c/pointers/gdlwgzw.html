// Subtracts the array's first-element pointer from its one-past-the-end pointer to get its length.
#include 
int main()
{
    int min, max;
    int arr[5] = {100, 15, 8, 45, 93};
    // &arr + 1 points one whole array past arr; pointer subtraction counts its elements.
    int length = *(&arr + 1) - arr;
    printf("Length of a given array is %d", length);
    return 0;
}
