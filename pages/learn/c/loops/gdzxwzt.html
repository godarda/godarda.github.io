C program to print the right half star pyramid pattern
{{ site.cfile }}
{%- highlight c -%}
// Prints a left-aligned increasing star triangle.
#include 
int main()
{
    int i, j;
    // The outer loop controls row width; the inner loop prints that many stars.
    for (i = 1; i <= 10; i++)
    {
        for (j = 1; j <= i; j++)
        {
            printf("*");
        }
        printf("\n");
    }
}

C program to print the left half star pyramid pattern
{{ site.cfile }}
{%- highlight c -%}
// Adds leading spaces to right-align an increasing star triangle.
#include 
int main()
{
    int i, j;
    for (i = 1; i <= 10; i++)
    {
        // Leading spaces shift rows right as the star count increases.
        for (j = 1; j <= 10 - i; j++)
            printf(" ");
        for (j = 1; j <= i; j++)
        {
            printf("*");
        }
        printf("\n");
    }
    return 0;
}

C program to print the inverted right half star pyramid pattern
{{ site.cfile }}
{%- highlight c -%}
// Prints a left-aligned decreasing star triangle.
#include 
int main()
{
    int i, j;
    // Decreasing the outer-loop bound shortens each successive row.
    for (i = 10; i >= 1; i--)
    {
        for (j = 1; j <= i; j++)
        {
            printf("*");
        }
        printf("\n");
    }
    return 0;
}
