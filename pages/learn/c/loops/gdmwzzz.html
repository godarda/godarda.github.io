// Extracts binary digits from right to left and accumulates their decimal place values.
#include 
int main()
{
    long n;
    int r, dn = 0, i = 1;
    printf("———————————————————————————————————————————");
    printf("\nProgram to convert a binary number to decimal");
    printf("\n———————————————————————————————————————————");
    printf("\nEnter the binary number ");
    scanf("%ld", &n);
    while (n != 0)
    {
        // Extract the rightmost binary digit and apply its current place value.
        r = n % 10;
        dn = dn + r * i;
        // Double the place value before processing the next digit.
        i = i * 2;
        n = n / 10;
    }
    printf("\nEquivalent decimal number is %d", dn);
    printf("\n———————————————————————————————————————————\n");
    return 0;
}
