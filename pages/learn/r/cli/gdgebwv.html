# Create a string
> "GoDarda"
[1] "GoDarda"

# Print a string
> print("Hello World")
[1] "Hello World"

# Print a string
> cat("Hello World\n")
Hello World

# Length of vector
> length("GoDarda")
[1] 1

# Number of characters
> nchar("GoDarda")
[1] 7

# Convert to uppercase
> toupper("GoDarda")
[1] "GODARDA"

# Convert to lowercase
> tolower("GODARDA")
[1] "godarda"

# Extract substring
> substring("GoDarda",3,7)
[1] "Darda"

# Concatenate strings
> paste("Go","Darda")
[1] "Go Darda"

# Concatenate strings with separator
> paste("Go","Darda",sep="")
[1] "GoDarda"

# Concatenate strings with separator
> paste("Go","Darda",sep="@@@")
[1] "Go@@@Darda"


# Format string with width
> format("GoDarda",width=20)
[1] "GoDarda             "

# Format string with justify
> format("GoDarda",width=20,justify="")
Error in match.arg(justify) :
  'arg' should be one of “left”, “right”, “centre”, “none”

# Format string with justify left
> format("GoDarda",width=20,justify="l")
[1] "GoDarda             "

# Format string with justify center
> format("GoDarda",width=20,justify="c")
[1] "      GoDarda       "

# Format string with justify right
> format("GoDarda",width=20,justify="r")
[1] "             GoDarda"

# Format string with justify none
> format("GoDarda",width=20,justify="n")
[1] "GoDarda"
