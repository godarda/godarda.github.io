>>> a=[1, 2, 3, 4, 5, 6]
# Slicing the list
>>> a[2:5]
[3, 4, 5]

>>> a[2:10]
[3, 4, 5, 6]

>>> a[:]
[1, 2, 3, 4, 5, 6]

>>> a
[1, 2, 3, 4, 5, 6]

>>> strings=['Thomson','1010011','999.99','A1D3T6Z']

# Accessing elements using index
>>> strings[0]
'Thomson'

>>> strings[3]
'A1D3T6Z'

# Negative indexing
>>> strings[-1]
'A1D3T6Z'

>>> strings[-4]
'Thomson'

# Length of the list
>>> len(strings)
4

# Modifying elements
>>> strings[0]='Neil'
>>> strings[0]
'Neil'

>>> strings[-4]
'Neil'

>>> strings
['Neil', '1010011', '999.99', 'A1D3T6Z']

# Deleting element
>>> del strings[3]
>>> strings
['Neil', '1010011', '999.99']

# Appending element
>>> strings.append('Edison')
>>> strings
['Neil', '1010011', '999.99', 'Edison']

# Reversing the list
>>> strings.reverse()
>>> strings
['Edison', '999.99', '1010011', 'Neil']

# Sorting the list
>>> strings.sort()
>>> strings
['1010011', '999.99', 'Edison', 'Neil']

# Inserting element
>>> strings.insert(2,'Linus')
>>> strings
['1010011', '999.99', 'Linus', 'Edison', 'Neil']

# Removing element
>>> strings.remove('999.99')
>>> strings
['1010011', 'Linus', 'Edison', 'Neil']

# Concatenation
>>> strings+[1,2,3,4]
['1010011', 'Linus', 'Edison', 'Neil', 1, 2, 3, 4]

>>> strings
['1010011', 'Linus', 'Edison', 'Neil']

# Convert to list
>>> list(strings)
['1010011', 'Linus', 'Edison', 'Neil']

>>> list('strings')
['s', 't', 'r', 'i', 'n', 'g', 's']

# Membership testing
>>> 'neil' in strings
False

>>> 'Neil' in strings
True

>>> 'Thomson' not in strings
True

# Pop element
>>> strings.pop(0)
'1010011'

>>> strings
['Linus', 'Edison', 'Neil']

# Delete element by index
>>> del strings[2]
>>> strings
['Linus', 'Edison']
