godarda@gd:~$ python3

>>> KV={'s1':'Linus','s2':'Thomson','s3':'Edison'}
>>> KV
{'s3': 'Edison', 's2': 'Thomson', 's1': 'Linus'}

# Accessing value using key
>>> KV['s2']
'Thomson'

# Length of dictionary
>>> len(KV)
3

# Max and Min keys
>>> max(KV)
's3'

>>> min(KV)
's1'

# Copying dictionary
>>> Dict=KV.copy()
>>> Dict
{'s1': 'Linus', 's2': 'Thomson', 's3': 'Edison'}

# Adding/Modifying elements
>>> KV['s4']='Neil'
>>> KV
{'s3': 'Edison', 's2': 'Thomson', 's1': 'Linus', 's4': 'Neil'}

# Updating dictionary
>>> KV.update(s5='John')
>>> KV
{'s1': 'Linus', 's2': 'Thomson', 's3': 'Edison', 's4': 'Neil', 's5': 'John'}

# Deleting element
>>> del KV['s5']
>>> KV
{'s1': 'Linus', 's2': 'Thomson', 's3': 'Edison', 's4': 'Neil'}

# Get keys
>>> KV.keys()
dict_keys(['s1', 's2', 's3', 's4'])

# Get values
>>> KV.values()
dict_values(['Linus', 'Thomson', 'Edison', 'Neil'])

# Get items (key-value pairs)
>>> KV.items()
dict_items([('s1', 'Linus'), ('s2', 'Thomson'), ('s3', 'Edison'), ('s4', 'Neil')])

# Iterating over items
>>> for(a,b) in KV.items():
...     print(b,a)
...
Linus s1
Thomson s2
Edison s3
Neil s4

# Iterating over values
>>> for(a) in KV.values():
...     print(a)
...
Linus
Thomson
Edison
Neil

# Clearing dictionary
>>> KV.clear()
>>> KV
{}
